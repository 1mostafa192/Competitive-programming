struct  DSU {

    vector<int>sizes;
    vector<int>parents;
    void inint(int n) {
        sizes.resize(n+1);
        parents.resize(n+1);
        for (int i=1;i<=n;i++)sizes[i]=1;
        for (int i=1;i<=n;i++)parents[i]=i;
    }

    int find_root(int u) {
        if (parents[u]==u) return u;
        return parents[u]=find_root(parents[u]);
    }
    bool marge(int u,int v) {
        int root_u=find_root(u);
        int root_v=find_root(v);
        if (root_u==root_v)return 0;

        if (sizes[root_u]>sizes[root_v]) swap(root_u,root_v);
        parents[root_u]=root_v;
        sizes[root_v]+=sizes[root_u];
        return 1;
    }
};

int main()
{

    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n,q;cin>>n>>q;
    DSU ds;
    ds.inint(n);
    while (q--) {
        int type,u,v;cin>>type>>u>>v;
        if (type==1) {
            ds.marge(u,v);
        }
